create table projekt(
id_projekt int PRIMARY KEY auto_increment,
nazwa_projektu varchar(150) not null,
data_rozpoczecia date not null,
data_zakonczenia date
)

alter table projekt
add menadzer_projektu int,
add foreign key (menadzer_projektu) references pracownicy(id_pracownicy) on delete set null;
